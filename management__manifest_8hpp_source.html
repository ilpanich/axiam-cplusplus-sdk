178// ---------------------------------------------------------------------------
179// §27.7's C++ declarative form: designated-initializer aggregate specs plus an
180// AXIAM_MANIFEST(...) macro.
181//
182// ManifestEntity is an aggregate whose every member carries a default member
183// initializer, which is what makes `.key = "x", .name = "y"` legal without
184// naming the members you do not care about — and, less obviously, what keeps
185// -Wextra quiet, since -Wmissing-field-initializers fires on an omitted member
186// only when that member has no default.
187//
188// The macros fix `.kind` and nothing else. They are sugar over the aggregate,
189// not a second way to build one: each expands to a ManifestEntity value and
190// AXIAM_MANIFEST expands to a Manifest holding them, so a manifest written this
191// way and one deserialized from configuration are the same value and go through
192// the same plan()/apply(). §27.7 requires exactly that — "a declarative form
193// that talks to the network itself is a second implementation of §27.6, and the
194// two will disagree."
195//
196// Designated initializers must appear in DECLARATION order, which for
197// ManifestEntity is: key, name, description, resource_type, action, is_global,
198// depends_on.
199//
200// const auto manifest = AXIAM_MANIFEST(
201// AXIAM_PERMISSION(.key = "read", .name = "documents:read", .action = "read"),
202// AXIAM_ROLE(.key = "editor", .name = "editor", .depends_on = "read"),
203// AXIAM_RESOURCE(.key = "root", .name = "documents", .resource_type = "folder"),
204// AXIAM_GROUP(.key = "editors", .name = "editors", .depends_on = "editor"));
205//
206// Unqualified in the axiam:: namespace on purpose: a macro has no namespace, so
207// the expansions below name their types fully instead.
208// ---------------------------------------------------------------------------
209
211#define AXIAM_RESOURCE(...) \
212 ::axiam::management::ManifestEntity { .kind = ::axiam::management::ManifestKind::Resource, __VA_ARGS__ }

215#define AXIAM_PERMISSION(...) \
216 ::axiam::management::ManifestEntity { .kind = ::axiam::management::ManifestKind::Permission, __VA_ARGS__ }

221
223#define AXIAM_GROUP(...) \
224 ::axiam::management::ManifestEntity { .kind = ::axiam::management::ManifestKind::Group, __VA_ARGS__ }

230#define AXIAM_MANIFEST(...) \
231 ::axiam::management::Manifest { { __VA_ARGS__ } }
