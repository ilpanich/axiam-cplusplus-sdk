1// §10 route-guard interface + §11 declarative authorization helpers.
2//
3// Framework-agnostic: the host adapter (Crow / Pistache / any HTTP server) pulls
4// the raw credential out of the request; turning it into an AxiamUser is the job
5// of axiam::TokenAuthenticator in <axiam/authenticator.hpp>, which is the
6// supported §10 verification path (signature + exp + nbf + tenant binding, fail
7// closed). Do NOT build an AxiamUser straight from
8// JwksVerifier::verify_signature_only_unchecked — that primitive validates the
9// signature only, so a guard fed from it accepts expired and cross-tenant tokens.
10//
11// The helpers here run strictly AFTER that identity exists and compose on top of
12// the client's check_access surface; they never re-implement token verification.

72inline void require_access(Client& client, const std::optional<AxiamUser>& user,
73 const std::string& action, const std::string& resource_id,
74 std::optional<std::string> scope = std::nullopt) {
75 const AxiamUser& u = require_auth(user);
76 if (resource_id.empty()) {
77 // §11.3: unresolvable resource id is a programming error (400).
78 throw std::invalid_argument("invalid_request: unresolved resource id");
79 }
80 AccessDecision decision;
81 try {
82 decision = client.check_access(action, resource_id, std::move(scope), u.user_id);
83 } catch (const AuthzError&) {
84 throw; // server said 403/409 → denied
85 } catch (const NetworkError&) {
86 // §11.5: fail closed on transport failure; never allow.
87 throw AuthzError("authz_unavailable");
88 }
89 if (!decision.allowed) {
90 throw AuthzError("authorization_denied");
91 }
92}

137inline void require_access(Client& client, const std::optional<AxiamUser>& user,
138 const std::string& action, const std::string& resource_id,
139 const UmaChallenger& challenger,
140 std::optional<std::string> scope = std::nullopt) {
141 try {
142 require_access(client, user, action, resource_id, std::move(scope));
143 return;
144 } catch (const AuthzError& denial) {
145 // Captured before the nested try: inside it, current_exception() would be
146 // the *minting* failure, and a bare `throw;` there would surface that
147 // instead of the denial — turning a 403 into whatever went wrong at the
148 // Protection API, which is exactly the escalation this must not do.
149 const std::exception_ptr original = std::current_exception();
150 std::string header;
151 try {
152 auto ticket = client.uma_request_ticket(
153 challenger.pat, {UmaRequestedPermission{resource_id, {action}}});
154 header = uma_challenge_header(challenger.realm, challenger.as_uri, ticket);
155 } catch (const AxiamError&) {
156 // Swallowed deliberately — see UmaChallenger. The denial stands on
157 // its own; only the sugar is lost.
158 std::rethrow_exception(original);
159 }
160 throw AuthzChallengeError(denial.what(), std::move(header), denial.action(),
161 denial.resource_id());
162 }
163}

179class AxiamGuard {
180public:
181 using Authenticator = std::function<std::optional<AxiamUser>(const Request&)>;
182
183 explicit AxiamGuard(Authenticator auth) : auth_(std::move(auth)) {}
184
185 AxiamUser operator()(const Request& request) const {
186 auto user = auth_(request);
187 return require_auth(user);
188 }
189
190private:
191 Authenticator auth_;
