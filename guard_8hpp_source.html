1// §10 route-guard interface + §11 declarative authorization helpers.
2//
3// Framework-agnostic: the host adapter (Crow / Pistache / any HTTP server) pulls
4// the raw credential out of the request; turning it into an AxiamUser is the job
5// of axiam::TokenAuthenticator in <axiam/authenticator.hpp>, which is the
6// supported §10 verification path (signature + exp + nbf + tenant binding, fail
7// closed). Do NOT build an AxiamUser straight from
8// JwksVerifier::verify_signature_only_unchecked — that primitive validates the
9// signature only, so a guard fed from it accepts expired and cross-tenant tokens.
10//
11// The helpers here run strictly AFTER that identity exists and compose on top of
12// the client's check_access surface; they never re-implement token verification.

73inline void require_access(Client& client, const std::optional<AxiamUser>& user,
74 const std::string& action, const std::string& resource_id,
75 std::optional<std::string> scope = std::nullopt) {
76 const AxiamUser& u = require_auth(user);
77 if (resource_id.empty()) {
78 // §11.3: unresolvable resource id is a programming error (400).
79 throw std::invalid_argument("invalid_request: unresolved resource id");
80 }
81 AccessDecision decision;
82 try {
83 decision = client.check_access(action, resource_id, std::move(scope), u.user_id);
84 } catch (const AuthzError&) {
85 throw; // server said 403/409 → denied
86 } catch (const NetworkError&) {
87 // §11.5: fail closed on transport failure; never allow.
88 throw AuthzError("authz_unavailable");
89 }
90 if (!decision.allowed) {
91 throw AuthzError("authorization_denied");
92 }
93}

138inline void require_access(Client& client, const std::optional<AxiamUser>& user,
139 const std::string& action, const std::string& resource_id,
140 const UmaChallenger& challenger,
141 std::optional<std::string> scope = std::nullopt) {
142 try {
143 require_access(client, user, action, resource_id, std::move(scope));
144 return;
145 } catch (const AuthzError& denial) {
146 // Captured before the nested try: inside it, current_exception() would be
147 // the *minting* failure, and a bare `throw;` there would surface that
148 // instead of the denial — turning a 403 into whatever went wrong at the
149 // Protection API, which is exactly the escalation this must not do.
150 const std::exception_ptr original = std::current_exception();
151 std::string header;
152 try {
153 auto ticket = client.uma_request_ticket(
154 challenger.pat, {UmaRequestedPermission{resource_id, {action}}});
155 header = uma_challenge_header(challenger.realm, challenger.as_uri, ticket);
156 } catch (const AxiamError&) {
157 // Swallowed deliberately — see UmaChallenger. The denial stands on
158 // its own; only the sugar is lost.
159 std::rethrow_exception(original);
160 }
161 throw AuthzChallengeError(denial.what(), std::move(header), denial.action(),
162 denial.resource_id());
163 }
164}

181inline void require_access(Client& client, const std::optional<AxiamUser>& user,
182 const std::string& action, const std::string& resource_id,
183 const std::optional<McpChallenges>& challenges,
184 std::optional<std::string> scope = std::nullopt) {
185 const AxiamUser& u = require_auth(user);
186 if (resource_id.empty()) {
187 throw std::invalid_argument("invalid_request: unresolved resource id");
188 }
189 AccessDecision decision;
190 try {
191 decision = client.check_access(action, resource_id, scope, u.user_id);
192 } catch (const AuthzError&) {
193 throw;
194 } catch (const NetworkError&) {
195 throw AuthzError("authz_unavailable");
196 }
197 if (decision.allowed) return;
198 if (challenges.has_value() && scope.has_value() && decision.reason_code.has_value() &&
199 *decision.reason_code == ReasonCode::kNoGrant) {
200 throw AuthzChallengeError("authorization_denied",
201 mcp_insufficient_scope_challenge(*challenges, *scope), action,
202 resource_id);
203 }
204 throw AuthzError("authorization_denied");
205}

225template <typename Request>
226class AxiamGuard {
227public:
228 using Authenticator = std::function<std::optional<AxiamUser>(const Request&)>;
229
236 using CredentialProbe = std::function<bool(const Request&)>;
237
238 explicit AxiamGuard(Authenticator auth) : auth_(std::move(auth)) {}
239
249 AxiamGuard(Authenticator auth, std::optional<McpChallenges> challenges,
250 CredentialProbe has_credential)
251 : auth_(std::move(auth)),
252 challenges_(std::move(challenges)),
253 has_credential_(std::move(has_credential)) {}
254
255 AxiamUser operator()(const Request& request) const {
256 auto user = auth_(request);
257 if (user.has_value()) return *user;
258 if (!challenges_.has_value()) return require_auth(user); // unconfigured: identical to before §28
259 const bool presented = has_credential_ && has_credential_(request);
260 throw AuthChallengeError("authentication_failed",
261 presented ? challenges_->invalid_token : challenges_->no_credential);
262 }
263
271 const std::optional<McpChallenges>& mcp_challenges() const noexcept { return challenges_; }
272
273private:
274 Authenticator auth_;
275 std::optional<McpChallenges> challenges_;
276 CredentialProbe has_credential_;
277};
